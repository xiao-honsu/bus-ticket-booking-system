# 🎨 Frontend Redesign: Booking Seat Map Integration

Based on the 3 API documentation files, this document outlines the complete frontend redesign for the seat selection feature.

## 📋 API Endpoints Overview

### Core APIs

1. **GET `/api/v1/trips/:tripId/seat-map`** - Get seat layout with real-time status
2. **POST `/api/v1/seat-locks/lock`** - Lock selected seats
3. **POST `/api/v1/seat-locks/release`** - Release locks
4. **POST `/api/v1/seat-locks/extend`** - Extend lock duration
5. **POST `/api/v1/seat-locks/check-availability`** - Check seat availability

### Key Features

- **SessionId Management**: Auto-generated by backend, must be saved
- **Lock Duration**: 10 minutes with auto-expire
- **Currency**: VND (Vietnamese Dong) - integer format
- **Polling**: Refresh seat map every 30-60 seconds
- **Countdown Timer**: Show time remaining before lock expires

### SeatLockTimer.tsx - New Component

```typescript
'use client';

import { useEffect, useState } from 'react';
import { useAppSelector, useAppDispatch } from '@/lib/hooks';
import { setLockExpiresAt } from '@/lib/store/features/booking/bookingSlice';
import { Card, CardContent } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Clock, AlertTriangle } from 'lucide-react';
import { toast } from 'sonner';

export default function SeatLockTimer() {
  const lockExpiresAt = useAppSelector((state) => state.booking.lockExpiresAt);
  const dispatch = useAppDispatch();
  const [timeRemaining, setTimeRemaining] = useState<number>(0);

  useEffect(() => {
    if (!lockExpiresAt) return;

    const updateTimer = () => {
      const now = new Date().getTime();
      const expiresTime = new Date(lockExpiresAt).getTime();
      const remaining = Math.max(0, Math.floor((expiresTime - now) / 1000));

      setTimeRemaining(remaining);

      // Show warning at 2 minutes
      if (remaining === 120) {
        toast.warning('Your seat lock will expire in 2 minutes!');
      }

      // Expired
      if (remaining === 0) {
        toast.error('Seat lock expired. Please select seats again.');
        dispatch(setLockExpiresAt(null));
      }
    };

    updateTimer();
    const interval = setInterval(updateTimer, 1000);

    return () => clearInterval(interval);
  }, [lockExpiresAt, dispatch]);

  if (!lockExpiresAt || timeRemaining === 0) return null;

  const minutes = Math.floor(timeRemaining / 60);
  const seconds = timeRemaining % 60;
  const isUrgent = timeRemaining < 180; // Less than 3 minutes

  return (
    <Card className={isUrgent ? 'border-red-500 bg-red-50' : 'border-yellow-500 bg-yellow-50'}>
      <CardContent className="p-4">
        <div className="flex items-center gap-3">
          {isUrgent ? (
            <AlertTriangle className="w-5 h-5 text-red-600" />
          ) : (
            <Clock className="w-5 h-5 text-yellow-600" />
          )}
          <div className="flex-1">
            <p className={`text-sm font-semibold ${isUrgent ? 'text-red-900' : 'text-yellow-900'}`}>
              Seat Lock Time Remaining
            </p>
            <p className={`text-2xl font-bold ${isUrgent ? 'text-red-600' : 'text-yellow-600'}`}>
              {minutes}:{seconds.toString().padStart(2, '0')}
            </p>
          </div>
        </div>
        {isUrgent && (
          <p className="text-xs text-red-700 mt-2">
            ⚠️ Please complete your booking soon or your seats will be released.
          </p>
        )}
      </CardContent>
    </Card>
  );
}
```

```typescript
// utils/currency.ts
export const formatVND = (amount: number): string => {
  return new Intl.NumberFormat('vi-VN', {
    style: 'currency',
    currency: 'VND',
  }).format(amount);
};

// Usage
formatVND(250000); // "₫250,000" or "250.000 ₫"
formatVND(1500000); // "₫1,500,000"

// Alternative: Custom format
export const formatVNDCustom = (amount: number): string => {
  return `${amount.toLocaleString('vi-VN')} ₫`;
};
```

### 2. Lock Expiration Handling

```typescript
// Countdown timer (10 minutes = 600 seconds)
useEffect(() => {
  const timer = setInterval(() => {
    const remaining = calculateRemaining(lockExpiresAt);

    if (remaining <= 0) {
      // Expired: redirect or show modal
      toast.error('Lock expired. Please select again.');
      router.push(`/trip-detail/${tripId}`);
    } else if (remaining === 120) {
      // Warning at 2 minutes
      toast.warning('Lock expires in 2 minutes!');
    }
  }, 1000);

  return () => clearInterval(timer);
}, [lockExpiresAt]);
```

### 3. Error Handling

```typescript
try {
  await lockSeats({ ... });
} catch (error) {
  if (error.message.includes('already locked')) {
    toast.error('Seats no longer available. Refreshing...');
    await refetch(); // Refresh seat map
  } else if (error.message.includes('already booked')) {
    toast.error('Seats have been booked by another user.');
    await refetch();
  } else {
    toast.error('Failed to lock seats. Please try again.');
  }
}
```

### 4. Cleanup on Exit

```typescript
// Release locks when leaving page
useEffect(() => {
  const handleBeforeUnload = () => {
    if (sessionId) {
      // Use sendBeacon for reliable cleanup
      const blob = new Blob([JSON.stringify({ sessionId })], { type: 'application/json' });
      navigator.sendBeacon('/api/v1/seat-locks/release', blob);
      localStorage.removeItem('seatlock_session');
    }
  };

  window.addEventListener('beforeunload', handleBeforeUnload);
  return () => window.removeEventListener('beforeunload', handleBeforeUnload);
}, [sessionId]);
```

### 5. Real-time Updates

